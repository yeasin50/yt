Task repository#

We have initialize databse and CRUD responsibility.

We have tasks table which contains id, title,description and createdAt all as text. More on supported datatype for sqlfite.

 1import '../models/task.dart';

 3import 'package:sqflite/sqflite.dart';

 5class TaskRepo {
 6  TaskRepo._(this._db);
 7  final Database _db; // comes from sqflite
 8
 9  static Future<TaskRepo> create() async {
10    final db = await openDatabase(
11      'app.db', //databse name
12      version: 1, //version
13      onCreate: (db, version) async {
14        await db.execute('''
15                      CREATE TABLE tasks(
16                          id TEXT PRIMARY KEY,
17                          title TEXT,
18                          description TEXT,
19                          createdAt TEXT
20                          )
21                      ''');
22      },
23    );
24
25    return TaskRepo._(db);
26  }
27 }

  7class TaskPage extends StatefulWidget {
  8  const TaskPage({super.key, required this.taskRepo});
  9
 10  final TaskRepo taskRepo;
 11  @override
 12  State<TaskPage> createState() => _TaskPageState();

 14
 15class _TaskPageState extends State<TaskPage> {
 16  TaskRepo get repo => widget.taskRepo;
 17
 18  late Future<List<Task>> tasks = repo.getTasks();
 19
 20  Future<void> onNewTaskSave(Task task) async {
 21    final result = await repo.addTask(task);
 22    if (result == null) return;
 23
 24    tasks = repo.getTasks();
 25    setState(() {});
 26  }
 27
 28  void removeTask(Task t) async {
 29    final result = await repo.deleteTask(t.id);
 30    if (result == 0) return;
 31
 32    tasks = repo.getTasks();
 33    setState(() {});
 34  }
 35
 36  void updateTask(Task t, bool? isChecked) async {
 37    final result = await repo.toggleTaskStatus(
 38      t.id,
 39      isChecked == true ? TaskStatus.compelete : TaskStatus.due,
 40    );
 41
 42    if (result == 0) return;
 43
 44    tasks = repo.getTasks();
 45    setState(() {});
 46  }
 47
 48  @override
 49  Widget build(BuildContext context) {
 50    return Scaffold(
 51      appBar: AppBar(title: Text("Task list")),
 52      body: Column(
 53        children: [
 54          Expanded(
 55            child: FutureBuilder<List<Task>>(
 56              future: tasks,
 57              builder: (context, asyncSnapshot) {
 58                if (asyncSnapshot.connectionState == ConnectionState.waiting) {
 59                  return Center(child: CircularProgressIndicator());
 60                }
 61
 62                if (asyncSnapshot.hasError) {
 63                  return Center(child: Text(asyncSnapshot.error.toString()));
 64                }
 65
 66                final items = asyncSnapshot.data ?? [];
 67
 68                return ListView(
 69                  children: [
 70                    if (items.isEmpty) Center(child: Text("No task found")),
 71                    ...items.map(
 72                      (e) => ListTile(
 73                        title: Text(e.title),
 74                        subtitle: Text(e.description),
 75                        trailing: IconButton(
 76                          icon: Icon(Icons.delete),
 77                          onPressed: () => removeTask(e),
 78                        ),
 79
 80                        leading: Checkbox(
 81                          value: e.status == TaskStatus.compelete,
 82                          onChanged: (v) {
 83                            updateTask(e, v);
 84                          },
 85                        ),
 86                      ),
 87                    ),
 88                  ],
 89                );
 90              },
 91            ),
